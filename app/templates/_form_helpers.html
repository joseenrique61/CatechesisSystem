{# templates/_form_helpers.html #} {# Macro para renderizar un campo de formulario individual con etiquetas, errores y clases de Bootstrap #} {% macro render_field(field, class='') %}
{% if field.type != 'BooleanField' and field.type != 'RadioField' and field.type != 'SubmitField' %} {{ field.label(class="form-label") }} {% endif %} {# Añadir clases de Bootstrap al campo y marcar como inválido si hay errores #} {% if field.type == 'BooleanField' %}
{{ field(class="form-check-input " + class, **kwargs) }} {{ field.label(class="form-check-label") }}
{% elif field.type == 'RadioField' %}
{% for subfield in field %}
{{ subfield(class="form-check-input") }} {{ subfield.label(class="form-check-label") }}
{% endfor %}
{% elif field.type == 'SubmitField' %} {{ field(class="btn " + class, **kwargs) }} {% else %} {{ field(class="form-control" + (" is-invalid" if field.errors else "") + " " + class, **kwargs) }} {% endif %} {# Mostrar errores de validación debajo del campo #} {% if field.errors %}
{% for error in field.errors %} {{ error }}
{% endfor %}
{% endif %}
{% endmacro %} {# Macro para renderizar una sección completa de un formulario anidado (FormField) #} {% macro render_form_section(form_section, title) %}
{% if title %} {{ title }} {% endif %} {# Iterar sobre todos los campos dentro del FormField #} {% for field in form_section %} {# Si el campo es otro FormField, llamamos recursivamente a esta macro #} {% if field.type == 'FormField' %} {{ render_form_section(field, field.label.text) }} {% else %} {{ render_field(field) }} {% endif %} {% endfor %}
{% endmacro %}
